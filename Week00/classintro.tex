\subsection{What are Programming Challenges?}
\begin{frame}{Homework: What are Programming Challenges?}

  It is a puzzle that you solve by writing a program.
  \bigskip

  The program reads the input, and must write the correct output.
  \begin{itemize}
  \item Difficulty 1: The output must be \alert{exactly} correct.
  \item Difficulty 2: The test input is \alert{not known}.
  \item Difficulty 3: There is a maximum \alert{execution time}.
  \end{itemize}\bigskip

  In general, a correct program will be small (< 200 lines).
\end{frame}

\subsection{Example}
\begin{frame}{Programming Challenge: Example}

  {\bf Problem Description}\\
  Given a list of $n$ integers ($x_1, x_2, x_3, \ldots, x_n$), you have to calculate the sum from $x_a$ to $x_b$, many times.
  \medskip

  {\bf Input Description}
  \begin{itemize}
    \item First you receive 2 integers in a line: $n, m$. ($1 \leq n, m \leq 10000$)
    \item Next you receive $n$ integers in a line: $x$. ($-10000 \leq x \leq 10000$)
    \item Next you receive $m$ lines with two integers: $a, b$. ($1 \leq a,b \leq n$)
  \end{itemize}
  \medskip

  {\bf Output Description}\\
  For each query, you must print the sum of numbers between $x_a$ and $x_b$ in a single line.

\end{frame}

\begin{frame}[fragile, t]{Programming Challenge: Input/Output Example}

  {\bf Input}
\begin{verbatim}
10 2
3 5 2 -1 10 -5 3 10 2 9
1 4
3 8
\end{verbatim}
  \bigskip

  {\bf Output}
\begin{verbatim}
9     # 3 + 5 + 2 + -1
19    # 2 + -1 + 10 + -5 + 3 + 10
\end{verbatim}

\end{frame}

\begin{frame}[fragile, t]{Programming Challenge: Simple solution}

{\smaller
\begin{verbatim}
int main() {
  int[10000] numbers;
  int n, m, a, b;
  int sum;

  cin >> n >> m;
  for (int i = 0; i < n; i++) { cin << numbers[i]; } # read input
  for (int i = 0; i < m; i++) {                      # read queries
    cin << a << b; 
    sum = 0;
    for (int j = a - 1; j < b; j++) {
      sum += numbers[a-1];
    }
    cout >> sum >> endl;                             # output
  }
}
\end{verbatim}}
\end{frame}

\begin{frame}[t]{Programming Challenge: Simple solutions, simple problems}

  \begin{itemize}
  \item Problem 1: What if \alert{$a > b$}?\medskip
  
  The problem description says "\alert{between a and b}" but does not say which one is bigger. If $a > b$, \alert{the simple program will give a wrong answer!}\medskip
  
  {\bf Solution}: Change the program: if $a > b$, then $a = t, a = b, b = t$.
  \bigskip

  
  \item Problem 2: What if \alert{m is big}?\medskip
  
  Each query costs $|a-b|$. If $m \equiv n$ and $|a-b| \equiv n$, then the program costs $n * n = 10000 * 10000 = 10^{8}$\medskip

  {\bf Solution}: Use \alert{memoization} to store the sums, and calculate the difference of the sums!

  \end{itemize}
\bigskip
\end{frame}

\begin{frame}[fragile,t]{Programming Challenge: Memoization Example}

  {\smaller
  \begin{verbatim}
  int main() {
    int[10000] numbers; int[10000] memo; int n, m, a, b; int sum;
    cin >> n >> m;

    cin >> numbers[0]; memo[0] = numbers[0];
    for (int i = 1; i < n; i++) { 
      cin << numbers[i]; 
      memo[i] = memo[i-1] + numbers[i]                 # memo table
    }

    for (int i = 0; i < m; i++) {                      # read queries
      cin << a << b; 
      if a > b { int t = a; a = b; b = t; }            # sort a and b
      cout << memo[b] - memo[a - 1] << endl;           # sum in O(1)
    }
  }
  \end{verbatim}}
\end{frame}
